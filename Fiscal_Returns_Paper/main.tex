% Appendix sections are lettered A--E (article's \appendix sets \thesection to \Alph{section} and

\renewcommand{\thesection}{\Alph{section}}

% \autoref{app:x} prints "Appendix A" from anywhere (hyperref types appendix sections by anchor).
% \autoref of an appendix subsection prints "Section A.1"; write Appendix~\ref{...} if "Appendix A.1" is wanted.
% Appendix tables and figures are numbered A1, A2, ..., B1, ...; counters reset in every appendix
% section. \theH... keeps hyperref anchors distinct from the main-text ones.

\counterwithin*{figure}{section}
\renewcommand{\thetable}{\thesection\arabic{table}}
\renewcommand{\thefigure}{\thesection\arabic{figure}}
\renewcommand{\theHtable}{app.\thesection.\arabic{table}}
\renewcommand{\theHfigure}{app.\thesection.\arabic{figure}}
